Number(current.amount || 0) : Number(current || 0); if (!Number.isFinite(currentAmount) || currentAmount < 0) { throw new Error("INVALID_BALANCE"); } return { amount: currentAmount + amount }; }); await requestRef.update({ credited: true, creditedAt: Date.now() }); return { success: true, message: "تمت إضافة الرصيد." }; } catch (error) { await requestRef.update({ status: "approved", creditError: true }); throw new HttpsError( "internal", "تم قبول الطلب لكن تعذرت إضافة الرصيد. يلزم فحص الطلب قبل إعادة المحاولة." ); } });
